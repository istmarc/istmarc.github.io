32 void solve(matrix<T> &A, vector<T> &b) {
33 if (_options._method == ls_method::qr) {
34 auto [q, r] = ::ten::linalg::qr(A);
35 ::ten::vector<T> z = ::ten::transposed(q) * b;
36 size_t n = b.size();
37 _x = ten::vector<T>({n});
38 ::ten::linalg::backward_subtitution(r, z, _x);
39 } else if (_options._method == ls_method::lu) {
40 auto [P, L, U] = ::ten::linalg::lu(A);
41 // Solve Lz = t using forward subtitution where z = Ux and t = P^T b
42 ::ten::vector<T> t = ::ten::transposed(P) * b;
43 size_t n = b.size();
44 auto z = ::ten::vector<T>({n});
45 ::ten::linalg::forward_subtitution(L, t, z);
46 // Solve Ux = z using backward subtitution
47 _x = ten::vector<T>({n});
48 ::ten::linalg::backward_subtitution(U, z, _x);
49 } else if (_options._method == ls_method::svd) {
50 auto [U, Sigma, Vt] = ::ten::linalg::svd(A);
51 size_t n = b.size();
52 // TODO Make invSigma = ten::fill<diagonal<T>>({n, n}, 1) / Sigma
53 ::ten::diagonal<T> invSigma({n, n});
54 for (size_t i = 0; i < n; i++) {
55 invSigma[i] = T(1) / Sigma[i];
56 }
57 // FIXME Make this work _x = ::ten::transposed(Vt) *
58 // ::ten::dense(invSigma) * ::ten::transposed(U) * b;
59 ::ten::matrix<T> m = ::ten::transposed(Vt) * ::ten::dense(invSigma) *
60 ::ten::transposed(U);
61 _x = m * b;
62 }
63 }
