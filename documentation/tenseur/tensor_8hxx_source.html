50template <size_t Rank> struct mdseq {
51 std::array<size_t, Rank> _start;
52 std::array<size_t, Rank> _end;
53
54 mdseq(auto... sequences)
55 requires(std::is_same_v<decltype(sequences...[0]), seq> &&
56 sizeof...(sequences) == Rank)
57 {
58 std::array<seq, Rank> arr{sequences...};
59 for (size_t i = 0; i < Rank; i++) {
60 _start[i] = arr[i]._start;
61 _end[i] = arr[i]._end;
62 }
63 }
64};
